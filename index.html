EnemyStateMachine.cs Pattern: Hierarchical State Machine · Zero Garbage Allocation 
// Máquina de Estados para IA Enemiga (Beers & Crystals)
public abstract class EnemyState
{
    protected readonly EnemyController enemy;
    protected readonly StateMachine stateMachine;

    public EnemyState(EnemyController enemy, StateMachine sm)
    {
        this.enemy = enemy;
        this.stateMachine = sm;
    }

    public virtual void Enter() { }
    public virtual void LogicUpdate() { }
    public virtual void PhysicsUpdate() { }
    public virtual void Exit() { }
}

// Estado de Persecución con predicción de trayectoria
public class ChaseState : EnemyState
{
    public override void LogicUpdate()
    {
        float distToPlayer = Vector3.Distance(enemy.transform.position, enemy.Target.position);

        if (distToPlayer <= enemy.AttackRadius)
        {
            stateMachine.ChangeState(enemy.AttackTelegraphState);
            return;
        }

        if (distToPlayer > enemy.LoseAggroRadius)
        {
            stateMachine.ChangeState(enemy.PatrolState);
            return;
        }

        enemy.NavAgent.SetDestination(enemy.Target.position);
    }
}
LiquidWobble.hlsl HLSL · Custom Function for Unity URP & VR 
// Cálculo del plano de corte de líquido con inercia física
void CalculateLiquidPlane_float(
    float3 worldPos,
    float3 vesselPos,
    float3 wobbleVector,
    float fillAmount,
    out float mask,
    out float3 liquidNormal
)
{
    // Vector normal de la superficie inclinada según aceleración
    float3 upVector = normalize(float3(0.0, 1.0, 0.0) + wobbleVector);
    liquidNormal = upVector;

    // Distancia vertical relativa al centro de la jarra
    float3 relPos = worldPos - vesselPos;
    float planeDistance = dot(relPos, upVector);

    // Borde afilado con antialiasing mediante fwidth
    float fillHeight = (fillAmount - 0.5) * 2.0;
    float delta = fwidth(planeDistance);
    mask = smoothstep(-delta, delta, fillHeight - planeDistance);
}
round_manager_device.verse UEFN · Verse Programming Language 
# Dispositivo de control de ronda y puntuación competitiva (UEFN)
using { /Fortnite.com/Devices }
using { /Verse.org/Simulation }

round_manager_device := class(creative_device):
    @editable EliminationTracker : elimination_feed_device = elimination_feed_device{}
    @editable RoundTimer : timer_device = timer_device{}

    var ActivePlayers : [agent]int = map{}

    OnBegin<override>()<suspends> : void =
        EliminationTracker.EliminatedEvent.Subscribe(OnPlayerEliminated)
        RoundTimer.SuccessEvent.Subscribe(OnRoundTimeExpired)
        InitializeRoundState()

    OnPlayerEliminated(Result : elimination_result) : void =
        if (EliminatingAgent := Result.EliminatingCharacter?.GetAgent[]):
            if (CurrentScore := ActivePlayers[EliminatingAgent]):
                if (set ActivePlayers[EliminatingAgent] = CurrentScore + 100):
                    CheckVictoryCondition(EliminatingAgent)
